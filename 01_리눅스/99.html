A-4. grep / sed / awk

grep

grep -n "패턴" f        # 줄번호
grep -c "패턴" f        # 개수
grep -v "패턴" f        # 반대
grep -i "패턴" f        # 대소문자 무시
grep -r "패턴" .        # 재귀
grep -E "a|b" f         # 확장 정규식
grep -o "mAP=[0-9.]*"   # 매칭 부분만
grep -A3 -B3 "패턴"     # 앞뒤 3줄
grep -l "패턴" *.txt    # 파일명만

sed

sed 's/a/b/g' f              # 전체 치환
sed -i.bak 's/a/b/g' f       # 백업 남기고 직접 수정 ★
sed -n '5,10p' f             # 5~10줄
sed '/패턴/d' f              # 줄 삭제
sed '1i 텍스트' f            # 첫 줄 앞 삽입
sed -E 's/([0-9]+)/[\1]/g'   # 그룹 참조

awk

awk '{print $1}' f
awk -F',' '{print $2}' f
awk 'NF!=5 {print FILENAME}' f
awk '$1==3 {c++} END {print c}' f
awk '{s+=$4*$5} END {printf "%.4f\n", s/NR}' f
awk 'NR==FNR{m[$1]=$2;next}{print m[$1]}' map.txt data.txt   # 조인
awk '{printf "%-20s %5d\n", $1, $2}' f                        # 서식

B. 트러블슈팅

B-1. 권한 관련

Permission denied — 스크립트 실행 시

ls -l script.sh          # x 권한 확인
chmod +x script.sh

Permission denied — 디렉토리 접근 시

ls -ld dir               # 디렉토리 x 권한 확인
chmod +x dir             # 진입 권한

Permission denied — pip install 시

# 원인: 시스템 파이썬에 설치 시도
conda activate vision-ai      # 반드시 가상환경 먼저
pip install pkg
# sudo pip install 는 절대 금지 ★

SSH Permissions 0644 for 'id_ed25519' are too open

chmod 600 ~/.ssh/id_ed25519

B-2. 파이썬 · conda

ImportError: libGL.so.1: cannot open shared object file

sudo apt install -y libgl1 libglib2.0-0
# 원인: OpenCV는 시스템 라이브러리를 필요로 함

conda: command not found

source ~/anaconda3/bin/activate
conda init bash
exec bash

conda activate 가 안 됨

conda init bash
source ~/.bashrc

환경을 activate 했는데 시스템 파이썬이 실행됨

which python             # 경로 확인
echo $PATH
hash -r                  # 셸의 명령 캐시 초기화

environment.yml 재현 실패

원인 해결
플랫폼 종속 빌드 정보 --no-builds 또는 --from-history 로 재생성
pip 패키지 누락 pip freeze > requirements.txt 별도 제공
채널 불일치 conda config --add channels conda-forge

CondaHTTPError / 다운로드 지연

conda config --set remote_read_timeout_secs 120
# 사내 프록시가 있으면 ~/.condarc 에 proxy_servers 설정

B-3. Git

fatal: not a git repository

pwd                      # 위치 확인
git rev-parse --show-toplevel    # 저장소 루트 찾기
git init                 # 새로 시작할 때만

커밋했는데 이름이 이상함

git config --global user.name "이름"
git config --global user.email "메일"
git commit --amend --reset-author      # 직전 커밋만 수정

reset --hard 로 커밋을 날렸다

git reflog               # 이동 기록에서 해시 확인
git reset --hard <해시>
# 기본 90일간 복구 가능

.gitignore 가 동작 안 함

git check-ignore -v <파일>       # 규칙 확인
git rm --cached <파일>           # 이미 추적 중이면 제외 필요
git rm -r --cached .             # 전체 재적용 (주의)
git add . && git commit -m "chore: gitignore 재적용"

한글 파일명이 \355\225\234 처럼 보임

git config --global core.quotepath false

push 시 rejected — non-fast-forward

git fetch origin
git log HEAD..origin/main        # 원격에 뭐가 있나
git pull --rebase                # 또는 merge
git push

충돌 해결 중 포기하고 싶다

git merge --abort
# rebase 중이면
git rebase --abort

fatal: refusing to merge unrelated histories

git pull origin main --allow-unrelated-histories
# 원인: 원격에 README를 만들면서 저장소를 생성한 경우

GitHub push 시 인증 실패

ssh -T git@github.com            # SSH 테스트
git remote -v                    # URL이 https인지 ssh인지
git remote set-url origin git@github.com:계정/저장소.git
# HTTPS면 비밀번호 대신 PAT 사용

100MB 초과 파일 push 거부

remote: error: File model.pt is 250 MB; this exceeds GitHub's file size limit

git rm --cached model.pt
echo "*.pt" >> .gitignore
git commit -m "chore: 모델 가중치 제외"
# 이력에서도 지우려면 git filter-repo 필요 (별도 학습)
# 권장: GitHub Releases 또는 별도 스토리지 사용

B-4. 셸 스크립트

bad interpreter: No such file or directory

# 또는
dos2unix script.sh

[: too many arguments

# 원인: 변수에 공백이 있는데 따옴표를 안 씀
if [ "$VAR" = "value" ]; then    # ✅ 따옴표 필수

unbound variable (set -u 사용 시)

VAR="${1:-기본값}"               # 기본값 지정
VAR="${VAR:-}"                   # 빈 값 허용

스크립트가 중간에 멈춤 (set -e)

bash -x script.sh                # 어디서 멈추는지 추적
# 실패해도 계속하려면
cmd || true

cron에서만 실패

원인 해결
PATH가 최소 스크립트에 export PATH=... 명시
상대경로 사용 절대경로로 변경
conda 환경 미활성화 source ~/anaconda3/etc/profile.d/conda.sh && conda activate env
환경변수 없음 .bashrc 를 source 하거나 직접 설정
로그 없음 >> /path/log 2>&1 추가

shellcheck 경고 대응

코드 의미 해결
SC2086 따옴표 없는 변수 "$VAR"
SC2046 따옴표 없는 명령 치환 "$(cmd)"
SC2164 cd 실패 미처리 cd dir \|\| exit
SC2181 $? 직접 검사 if cmd; then

D. 자주 쓰는 one-liner 모음 (Vision AI 실무)

# 이미지 개수
find . -name "*.jpg" | wc -l

# 라벨 없는 이미지
comm -23 <(ls *.jpg | sed 's/\.jpg$//' | sort) \
         <(ls *.txt | sed 's/\.txt$//' | sort)

# 클래스 분포
cat *.txt | cut -d' ' -f1 | sort | uniq -c | sort -rn

# 빈 라벨 파일
find . -name "*.txt" -size 0

# 필드 개수 오류
awk 'NF!=5 && NF!=0 {print FILENAME}' *.txt | sort -u

# 좌표 범위 초과
awk '{for(i=2;i<=5;i++) if($i<0||$i>1){print FILENAME": "$0;next}}' *.txt

# bbox 평균 면적
awk '{s+=$4*$5} END {printf "%.5f\n", s/NR}' *.txt

# 초소형 객체 개수 (면적 1% 미만)
awk '$4*$5<0.01 {c++} END {print c+0}' *.txt

# YOLO → 픽셀 좌표
awk -v W=4848 -v H=2704 '{printf "%d %.0f %.0f %.0f %.0f\n", $1,
  ($2-$4/2)*W, ($3-$5/2)*H, ($2+$4/2)*W, ($3+$5/2)*H}' label.txt

# 전체 라벨을 CSV로
awk '{split(FILENAME,a,".");print a[1]","$1","$2","$3","$4","$5}' *.txt

# 학습 로그에서 mAP 추출
grep -o "mAP50=[0-9.]*" train.log | cut -d= -f2

# 목표 성능(0.7) 달성 시점
awk -F'mAP50=' '/mAP50=/ && $2+0>=0.7 {print; exit}' train.log

# 파일명 공백 제거
find . -name "* *" -type f | while read -r f; do mv "$f" "${f// /_}"; done

# 폴더별 용량 큰 순
du -sh */ | sort -rh | head

# 깨진 심볼릭 링크
find . -xtype l

# 최근 수정 파일 10개
find . -type f -printf '%T@ %p\n' | sort -rn | head -10 | cut -d' ' -f2-

# 저장소에 금지 파일이 있는지
git ls-files | grep -Ei '\.(jpg|jpeg|png|bmp|mp4|pt|pth|onnx)$'

# 커밋 통계
git log --oneline | wc -l
git log --pretty='%an' | sort | uniq -c | sort -rn
